// 定义函数
FSTAR := function(F)
    G, phi := MultiplicativeGroup(F);
    res := [ phi(x) : x in G ];
    return res;
end function;

// 1. 定义一个有限域 F，例如 GF(2^3) = GF(8)
F := GF(2, 3);

// 2. 获取乘法群
// G 是 F 的乘法群，phi 是从 G 到 F* 的映射
G, phi := MultiplicativeGroup(F);

// 3. 查看乘法群的结构
print "MultiplicativeGroup(F):", G;

// 4. 获取乘法群的生成元，并映射回有限域中的实际元素
// 乘法群的生成元通常是 G.1
generator_in_F := phi(G.1);
print "generator in F", generator_in_F;

// 5. 验证它是否确实生成了所有的非零元素
print "all elements in MultiplicativeGroup(F):";
for x in G do
    print phi(x);
end for;

FSTAR(F);
